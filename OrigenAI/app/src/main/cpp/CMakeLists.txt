cmake_minimum_required(VERSION 3.22.1)
project(origenllama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED ORIGEN_LLAMA_SOURCE)
  message(FATAL_ERROR "Set ORIGEN_LLAMA_SOURCE to the pinned llama.cpp checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${ORIGEN_LLAMA_SOURCE}" "${CMAKE_BINARY_DIR}/llama-vendor" EXCLUDE_FROM_ALL)
add_library(origenllama SHARED origen_llama.cpp)
target_include_directories(origenllama PRIVATE "${ORIGEN_LLAMA_SOURCE}/include")
target_link_libraries(origenllama PRIVATE llama log android)
target_compile_options(origenllama PRIVATE -O2 -fvisibility=hidden)
